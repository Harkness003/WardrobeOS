+/// Centralizes incremental evidence precedence; widgets only render its result.
+class GarmentAnalysisMergeEngine {
+  static const sourcePriority = <String, int>{
+    'default': 0, 'ai_estimate': 1, 'main_photo': 2,
+    'targeted_photo': 3, 'label': 4, 'user': 5,
+  };
+
+  const GarmentAnalysisMergeEngine();

+  GarmentAnalysisResult merge({
+    GarmentAnalysisResult? previous,
+    required GarmentAnalysisResult incoming,
+    Map<String, String> userCorrections = const {},
+  }) {
+    if (previous == null && userCorrections.isEmpty) return incoming;
+    final oldSources = previous?.fieldSources ?? const <String, String>{};
+    final newSources = incoming.fieldSources ?? const <String, String>{};
+    final sources = <String, String>{...oldSources};
+    final statuses = <String, String>{...?previous?.fieldStatuses};
+    final explanations = <String, String>{...?previous?.fieldExplanations};
+    final confidences = <String, double>{...?previous?.fieldConfidences};
+    final divergences = <String, Object?>{};
+
+    String? select(String field, String? oldValue, String? newValue) {
+      final correction = userCorrections[field]?.trim();
+      if (correction != null && correction.isNotEmpty) {
+        if (newValue != null && newValue != correction) divergences[field] = newValue;
+        sources[field] = 'user'; statuses[field] = 'corrected'; confidences[field] = 1;
+        return correction;
+      }
+      if (newValue == null || newValue.trim().isEmpty) return oldValue;
+      if (oldValue == null || oldValue.trim().isEmpty ||
+          _rank(newSources[field]) >= _rank(oldSources[field])) {
+        sources[field] = newSources[field] ?? 'ai_estimate';
+        statuses[field] = incoming.fieldStatuses?[field] ?? statuses[field] ?? 'estimated';
+        explanations[field] = incoming.fieldExplanations?[field] ?? explanations[field] ?? '';
+        confidences[field] = incoming.fieldConfidences[field] ?? confidences[field] ?? incoming.globalConfidence;
+        return newValue;
+      }
+      return oldValue;
+    }
+
+    final metadata = <String, Object?>{...?previous?.analysisMetadata, ...?incoming.analysisMetadata};
+    if (divergences.isNotEmpty) metadata['userCorrectionDivergences'] = divergences;
+    final old = previous;
+    return GarmentAnalysisResult(
+      isUsableImage: incoming.isUsableImage,
+      rejectionReason: incoming.rejectionReason,
+      suggestedName: select('suggestedName', old?.suggestedName, incoming.suggestedName),
+      category: select('category', old?.category, incoming.category),
+      preciseType: select('preciseType', old?.preciseType, incoming.preciseType),
+      primaryColor: select('primaryColor', old?.primaryColor, incoming.primaryColor),
+      material: select('material', old?.material, incoming.material),
+      season: select('season', old?.season, incoming.season),
+      visibleBrand: select('visibleBrand', old?.visibleBrand, incoming.visibleBrand),
+      globalConfidence: incoming.globalConfidence >= (old?.globalConfidence ?? 0) ? incoming.globalConfidence : old?.globalConfidence ?? 0,
+      imageQualityConfidence: incoming.imageQualityConfidence,
+      isBlurry: incoming.isBlurry, isTooDark: incoming.isTooDark,
+      isOverexposed: incoming.isOverexposed, garmentIsPartiallyHidden: incoming.garmentIsPartiallyHidden,
+      garmentIsTooSmall: incoming.garmentIsTooSmall, multipleMainGarments: incoming.multipleMainGarments,
+      backgroundIsProblematic: incoming.backgroundIsProblematic,
+      imageQualityWarnings: incoming.imageQualityWarnings,
+      fieldConfidences: confidences, fieldStatuses: statuses, fieldSources: sources,
+      fieldMetadata: {...?old?.fieldMetadata, ...?incoming.fieldMetadata},
+      overallConfidence: incoming.overallConfidence ?? old?.overallConfidence,
+      analysisMetadata: metadata, fieldExplanations: explanations,
+      warnings: {...?old?.warnings, ...incoming.warnings}.toList(),
+      styleSummary: incoming.styleSummary ?? old?.styleSummary,
+      styleStrengths: incoming.styleStrengths.isEmpty ? old?.styleStrengths ?? const [] : incoming.styleStrengths,
+      styleWeaknesses: incoming.styleWeaknesses.isEmpty ? old?.styleWeaknesses ?? const [] : incoming.styleWeaknesses,
+      styleAdvice: incoming.styleAdvice.isEmpty ? old?.styleAdvice ?? const [] : incoming.styleAdvice,
+      compatibleColors: incoming.compatibleColors.isEmpty ? old?.compatibleColors ?? const [] : incoming.compatibleColors,
+      lessSuitableColors: incoming.lessSuitableColors.isEmpty ? old?.lessSuitableColors ?? const [] : incoming.lessSuitableColors,
+      compatibleBottoms: incoming.compatibleBottoms.isEmpty ? old?.compatibleBottoms ?? const [] : incoming.compatibleBottoms,
+      compatibleShoes: incoming.compatibleShoes.isEmpty ? old?.compatibleShoes ?? const [] : incoming.compatibleShoes,

+      discouragedOccasions: incoming.discouragedOccasions.isEmpty ? old?.discouragedOccasions ?? const [] : incoming.discouragedOccasions,
+      versatilityExplanation: incoming.versatilityExplanation ?? old?.versatilityExplanation,
