+import 'package:wardrobeos/models/garment.dart';
+
+void main() {
+  test('normalise les formats objet et liste de fieldConfidences', () {
+    final object = GarmentAnalysisResult.fromJson({'isUsableImage': true, 'globalConfidence': .8, 'fieldConfidences': {'category': 1.4, 'bad': 'x'}});
+    final list = GarmentAnalysisResult.fromJson({'isUsableImage': true, 'fieldConfidences': [{'field': 'category', 'confidence': -.2}, {'field': 'bad', 'confidence': null}]});
+    expect(object.fieldConfidences, {'category': 1});
+    expect(list.fieldConfidences, {'category': 0});
+  });
+
+  test('accepte une réponse complète, partielle ou inutilisable', () {
+    final complete = GarmentAnalysisResult.fromJson({'isUsableImage': true, 'identification': {'suggestedName': ' Veste ', 'category': 'Vestes'}, 'style': {'formalityScore': 12}, 'climate': {'estimatedMinTemperature': 20, 'estimatedMaxTemperature': 5}, 'globalConfidence': 2});
+    final validator = GarmentAnalysisValidator(categoryNormalizer: const GarmentValueNormalizer(['Vestes']), colorNormalizer: const GarmentValueNormalizer([]), materialNormalizer: const GarmentValueNormalizer([]), seasonNormalizer: const GarmentValueNormalizer([]));
+    final cleaned = validator.validate(complete).analysis;
+    expect(cleaned.suggestedName, 'Veste');
+    expect(cleaned.style['formalityScore'], 10);
+    expect(cleaned.climate['estimatedMinTemperature'], 5);
+    expect(cleaned.globalConfidence, 1);
+    expect(GarmentAnalysisResult.fromJson({'isUsableImage': true}).identification, isEmpty);
+    final rejected = GarmentAnalysisResult.fromJson({'isUsableImage': false, 'rejectionReason': 'Image trop sombre'});
+    expect(rejected.rejectionReason, 'Image trop sombre');
+  });
+
+  test('préserve inconnues et groupes lors du round-trip', () {
+    final source = GarmentAnalysisResult.fromJson({'isUsableImage': true, 'materials': {'estimatedComposition': 'Inconnue'}, 'nonVisibleFields': ['étiquette'], 'stylisticAnalysis': {'honestVerdict': 'Usage limité'}});
+    final restored = GarmentAnalysisResult.fromJson(source.toJson());
+    expect(restored.materials['estimatedComposition'], 'Inconnue');
+    expect(restored.nonVisibleFields, ['étiquette']);
+    expect(restored.stylisticAnalysis['honestVerdict'], 'Usage limité');
+  });
+
+  test('un ancien vêtement sans analyse Vision reste chargeable', () {
+    final old = Garment.fromMap({'id':'1','name':'Chemise','category':'Hauts','wear_count':0,'is_favorite':0,'created_at':'2026-01-01T00:00:00.000','updated_at':'2026-01-01T00:00:00.000'});
+    expect(old.visionAnalysis, isNull);
+  });
+}
